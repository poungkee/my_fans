```typescript
// daumJsonParser.ts
async function crawlDaumNews(category: string) {
  // 1. Daum News API 호출
  const response = await axios.get(
    `https://news.daum.net/api/category/${category}`
  );

  const articles = response.data.items;

  // 2. 각 기사별 상세 정보 수집
  for (const article of articles) {
    // 기사 상세 페이지 접속
    const detailResponse = await axios.get(article.url);

    // Cheerio로 HTML 파싱
    const $ = cheerio.load(detailResponse.data);

    // 본문 추출
    const content = $('div.article_view').text().trim();

    // 기자명 추출
    const journalist = $('span.txt_info').text().trim();

    // DB 저장
    await saveToDatabase({
      title: article.title,
      content: content,
      url: article.url,
      image_url: article.image,
      original_source: article.source,
      original_category: category,
      journalist: journalist,
      pub_date: new Date(article.pubDate)
    });

    // 요청 간격 (1초 대기)
    await delay(1000);
  }
}
```

```typescript
export async function processRawNews() {
  // 1. 미처리 기사 조회
  const rawNews = await pool.query(`
    SELECT * FROM raw_news_articles
    WHERE processed = FALSE
    LIMIT ${BATCH_SIZE}
  `);

  for (const news of rawNews.rows) {
    // 2. 분류 API 호출
    const classification = await axios.post(
      'http://classification-api:5000/classify',
      {
        title: news.title,
        content: news.content,
        original_source: news.original_source,
        original_category: news.original_category
      }
    );

    // 3. news_articles에 저장
    await pool.query(`
      INSERT INTO news_articles (
        title, content, url, image_url,
        source_id, category_id, journalist, pub_date
      )
      VALUES ($1, $2, $3, $4, $5, $6, $7, $8)
      ON CONFLICT (url) DO NOTHING
    `, [
      news.title,
      news.content,
      news.url,
      news.image_url,
      classification.data.source_id,
      classification.data.category_id,
      news.journalist,
      news.pub_date
    ]);

    // 4. processed 플래그 업데이트
    await pool.query(`
      UPDATE raw_news_articles
      SET processed = TRUE
      WHERE id = $1
    `, [news.id]);
  }

  console.log(`✅ ${rawNews.rows.length}개 기사 처리 완료`);
}
```

```typescript
  const articles = await pool.query(`

    FROM news_articles
    WHERE ai_summary IS NULL

      SET ai_summary = $1
      WHERE id = $2
```

```typescript
export async function extractKeywords() {
  const articles = await pool.query(`
    SELECT id, title, content
    FROM news_articles
    WHERE NOT EXISTS (
      SELECT 1 FROM news_keywords WHERE news_id = news_articles.id
    )
    LIMIT 50
  `);

  for (const article of articles.rows) {
    // KoNLPy 또는 간단한 형태소 분석
    const keywords = await extractKeywordsFromText(
      article.title + ' ' + article.content
    );

    for (const keyword of keywords) {
      // keywords 테이블에 삽입 (또는 빈도 증가)
      const keywordResult = await pool.query(`
        INSERT INTO keywords (keyword)
        VALUES ($1)
        ON CONFLICT (keyword) DO UPDATE SET
          frequency = keywords.frequency + 1
        RETURNING id
      `, [keyword.word]);

      // news_keywords 관계 저장
      await pool.query(`
        INSERT INTO news_keywords (news_id, keyword_id, relevance)
        VALUES ($1, $2, $3)
        ON CONFLICT DO NOTHING
      `, [article.id, keywordResult.rows[0].id, keyword.score]);
    }
  }

  console.log(`✅ ${articles.rows.length}개 키워드 추출 완료`);
}
```

```typescript
export async function analyzeBias() {

    SELECT id, title, content
    FROM news_articles

    )

    // 편향성 분석 AI 호출
```

```python
# summarizer.py
from transformers import T5Tokenizer, T5ForConditionalGeneration

class NewsSummarizer:
    def __init__(self):
        model_name = os.getenv('MODEL_NAME', 'eenzeenee/t5-base-korean-summarization')
        self.tokenizer = T5Tokenizer.from_pretrained(model_name)
        self.model = T5ForConditionalGeneration.from_pretrained(model_name)

    def summarize(self, text: str, max_length: int = 100):
        # 토큰화
        inputs = self.tokenizer(
            text,
            return_tensors="pt",
            max_length=512,
            truncation=True
        )

        # 요약 생성
        summary_ids = self.model.generate(
            inputs.input_ids,
            max_length=max_length,
            min_length=30,
            length_penalty=2.0,
            num_beams=4,
            early_stopping=True
        )

        # 디코딩
        summary = self.tokenizer.decode(
            summary_ids[0],
            skip_special_tokens=True
        )

        return summary
```

#### 감성 분석 로직

```python
# sentiment_analyzer.py
from transformers import BertTokenizer, BertForSequenceClassification
import torch

class SentimentAnalyzer:
    def __init__(self):
        # KoBERT 모델 로드
        self.tokenizer = BertTokenizer.from_pretrained('monologg/kobert')
        self.model = BertForSequenceClassification.from_pretrained(
            'monologg/kobert',
            num_labels=3  # 긍정/중립/부정
        )

    def analyze(self, text: str):
        # 토큰화
        inputs = self.tokenizer(
            text,
            return_tensors='pt',
            truncation=True,
            max_length=512
        )

        # 추론
        with torch.no_grad():
            outputs = self.model(**inputs)
            logits = outputs.logits
            probs = torch.softmax(logits, dim=1)

        # 결과 추출
        sentiment_idx = torch.argmax(probs, dim=1).item()
        confidence = probs[0][sentiment_idx].item()

        sentiments = ['negative', 'neutral', 'positive']
        sentiment = sentiments[sentiment_idx]

        # 편향 점수 계산 (-1.0 ~ +1.0)
        score = probs[0][2].item() - probs[0][0].item()

        return {
            'sentiment': sentiment,
            'confidence': confidence,
            'score': score
        }
```
