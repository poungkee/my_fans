```typescript
export const authMiddleware = async (req, res, next) => {
  const token = req.headers.authorization?.replace('Bearer ', '');

  if (!token) {
    return res.status(401).json({ message: 'Unauthorized' });
```

**개선된 구현**:

```typescript
// 1. Refresh Token 추가 (Access Token 유효기간 단축)
interface TokenPair {
  accessToken: string;      // 15분 만료
  refreshToken: string;     // 7일 만료
  expiresIn: number;
}

export async function issueTokens(userId: number): Promise<TokenPair> {
  const accessToken = jwt.sign(
    { userId },
    process.env.JWT_SECRET,
    { expiresIn: '15m', issuer: 'fans-api', audience: 'fans-app' }
  );

  const refreshToken = jwt.sign(
    { userId, type: 'refresh' },
    process.env.JWT_REFRESH_SECRET,
    { expiresIn: '7d', issuer: 'fans-api' }
  );

  // 토큰 저장 (로그아웃 추적용)
  await redis.setex(`token:${userId}:${refreshToken}`, 604800, '1'); // 7일

  return {
    accessToken,
    refreshToken,
    expiresIn: 900  // 15분 (초 단위)
  };
}

// 2. Token Rotation (Refresh Token 재발급)
export async function refreshAccessToken(refreshToken: string): Promise<TokenPair> {
  try {
    const decoded = jwt.verify(refreshToken, process.env.JWT_REFRESH_SECRET);

    // 이전 토큰 무효화 (Rotation)
    await redis.del(`token:${decoded.userId}:${refreshToken}`);

    // 새 토큰 발급
    return issueTokens(decoded.userId);
  } catch (error) {
    throw new Error('Invalid or expired refresh token');
  }
}

// 3. Token Blacklist (로그아웃 시)
export async function logout(accessToken: string, refreshToken: string): Promise<void> {
  const decoded = jwt.decode(accessToken);
  const expiresIn = Math.ceil((decoded.exp - Math.floor(Date.now() / 1000)) * 1000);

  // Access Token 블랙리스트 추가
  await redis.setex(`blacklist:${accessToken}`, Math.max(expiresIn, 1000), '1');

  // Refresh Token 삭제
  const refreshDecoded = jwt.decode(refreshToken);
  await redis.del(`token:${refreshDecoded.userId}:${refreshToken}`);
}

// 4. 강화된 토큰 검증
export async function verifyToken(token: string): Promise<any> {
  try {
    // 블랙리스트 확인
    const isBlacklisted = await redis.exists(`blacklist:${token}`);
    if (isBlacklisted) {
      throw new Error('Token is blacklisted');
    }

    // 토큰 검증
    const decoded = jwt.verify(token, process.env.JWT_SECRET);

    // 발급자 및 대상자 확인
    if (decoded.issuer !== 'fans-api' || decoded.audience !== 'fans-app') {
      throw new Error('Invalid token issuer or audience');
    }

    return decoded;
  } catch (error) {
    throw new Error(`Token verification failed: ${error.message}`);
  }
}
```

#### OAuth 2.0 보안 (PKCE 적용)

```typescript
// PKCE (Proof Key for Code Exchange) - Authorization Code 탈취 방지
import crypto from 'crypto';

// 1. 인증 요청 시작 (클라이언트)
export async function initiateOAuthLogin(req, res) {
  // Code Verifier 생성 (무작위 문자열)
  const codeVerifier = crypto.randomBytes(32).toString('base64url');

  // Code Challenge 생성 (SHA256 해시)
  const codeChallenge = crypto
    .createHash('sha256')
    .update(codeVerifier)
    .digest('base64url');

  // 세션에 저장 (CSRF 토큰 역할도 함)
  req.session.oauthState = crypto.randomBytes(16).toString('hex');
  req.session.codeVerifier = codeVerifier;

  // 인증 요청 생성
  const authUrl = `https://kauth.kakao.com/oauth/authorize?${new URLSearchParams({
    client_id: process.env.KAKAO_CLIENT_ID,
    redirect_uri: `${process.env.API_BASE_URL}/auth/callback`,
    response_type: 'code',
    state: req.session.oauthState,
    code_challenge: codeChallenge,
    code_challenge_method: 'S256',  // SHA256 방식 사용
  })}`;

  res.redirect(authUrl);
}

// 2. 콜백 처리 (토큰 요청)
export async function oauthCallback(req, res) {
  const { code, state } = req.query;

  // CSRF 검증 (state 확인)
  if (state !== req.session.oauthState) {
    return res.status(400).json({ error: 'Invalid state parameter' });
  }

  // Code Verifier와 함께 토큰 요청
  const tokenResponse = await axios.post('https://kauth.kakao.com/oauth/token', {
    grant_type: 'authorization_code',
    client_id: process.env.KAKAO_CLIENT_ID,
    client_secret: process.env.KAKAO_CLIENT_SECRET,
    redirect_uri: `${process.env.API_BASE_URL}/auth/callback`,
    code,
    code_verifier: req.session.codeVerifier,  // PKCE 완성
  });

  // 토큰으로 사용자 정보 조회 및 로그인
  const userInfo = await getUserInfoFromKakao(tokenResponse.access_token);
  const tokens = await issueTokens(userInfo.id);

  // 세션 정리
  delete req.session.codeVerifier;
  delete req.session.oauthState;

  return res.json(tokens);
}
```

#### RBAC (Role-Based Access Control)

```typescript
// 1. 역할 정의
// backend/api/src/entities/User.entity.ts
export enum UserRole {
  USER = 'user',           // 일반 사용자
  MODERATOR = 'moderator', // 콘텐츠 관리자
  ADMIN = 'admin',         // 시스템 관리자
}

export enum Permission {
  // 사용자 권한
  READ_ARTICLES = 'read:articles',
  CREATE_BOOKMARK = 'create:bookmark',

  // 관리자 권한
  DELETE_ARTICLES = 'delete:articles',
  MANAGE_USERS = 'manage:users',
  VIEW_ANALYTICS = 'view:analytics',
}

@Entity('users')
export class User {
  @PrimaryGeneratedColumn()
  id: number;

  @Column({
    type: 'enum',
    enum: UserRole,
    default: UserRole.USER,
  })
  role: UserRole;

  // 역할별 권한 매핑
  getPermissions(): Permission[] {
    const rolePermissions: Record<UserRole, Permission[]> = {
      [UserRole.USER]: [
        Permission.READ_ARTICLES,
        Permission.CREATE_BOOKMARK,
      ],
      [UserRole.MODERATOR]: [
        Permission.READ_ARTICLES,
        Permission.CREATE_BOOKMARK,
        Permission.DELETE_ARTICLES,
      ],
      [UserRole.ADMIN]: [
        Permission.READ_ARTICLES,
        Permission.DELETE_ARTICLES,
        Permission.MANAGE_USERS,
        Permission.VIEW_ANALYTICS,
      ],
    };

    return rolePermissions[this.role] || [];
  }
}

// 2. RBAC 미들웨어
// backend/api/src/middleware/rbac.middleware.ts
export const requireRole = (...allowedRoles: UserRole[]) => {
  return (req, res, next) => {
    if (!req.user) {
      return res.status(401).json({ message: 'Unauthorized' });
    }

    if (!allowedRoles.includes(req.user.role)) {
      return res.status(403).json({
        message: 'Insufficient permissions',
        required_roles: allowedRoles,
        user_role: req.user.role
      });
    }

    next();
  };
};

// 권한 기반 접근 제어
export const requirePermission = (requiredPermission: Permission) => {
  return async (req, res, next) => {
    if (!req.user) {
      return res.status(401).json({ message: 'Unauthorized' });
    }

    const user = await userRepository.findOne({
      where: { id: req.user.userId }
    });

    const permissions = user.getPermissions();
    if (!permissions.includes(requiredPermission)) {
      return res.status(403).json({
        message: 'Permission denied',
        required_permission: requiredPermission,
        user_permissions: permissions
      });
    }

    next();
  };
};

// 3. 사용 예시
app.delete('/api/news/:id',
  authMiddleware,
  requireRole(UserRole.ADMIN, UserRole.MODERATOR),
  requirePermission(Permission.DELETE_ARTICLES),
  deleteNewsController
);
```

```typescript
app.post('/auth/login', async (req, res) => {

  const { refreshToken } = req.body;

    res.status(401).json({ error: 'Invalid refresh token' });
  }
```

```typescript
  // 프로덕션 환경에서 HTTP 요청을 HTTPS로 리다이렉트
```

```sql
-- PostgreSQL pgcrypto 확장 활성화
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- 암호화 함수 (AES-256)
CREATE OR REPLACE FUNCTION encrypt_aes(text_data TEXT, key TEXT)
RETURNS TEXT AS $$
BEGIN
  RETURN encode(
    pgcrypto.encrypt(text_data::bytea, key::bytea, 'aes'),
    'base64'
  );
END;
$$ LANGUAGE plpgsql;

-- 복호화 함수
CREATE OR REPLACE FUNCTION decrypt_aes(encrypted_data TEXT, key TEXT)
RETURNS TEXT AS $$
BEGIN
  RETURN convert_from(
    pgcrypto.decrypt(decode(encrypted_data, 'base64'), key::bytea, 'aes'),
    'UTF8'
  );
END;
$$ LANGUAGE plpgsql;

-- 민감 정보 암호화 (이메일, 전화번호 등)
ALTER TABLE users ADD COLUMN email_encrypted TEXT;

UPDATE users
SET email_encrypted = encrypt_aes(email, 'encryption_key_from_env')
WHERE email_encrypted IS NULL;
```

```typescript
// backend/api/src/entities/User.entity.ts
import crypto from 'crypto';

export class UserEncryption {
  private static readonly algorithm = 'aes-256-gcm';
  private static readonly encryptionKey = process.env.ENCRYPTION_KEY;

  static encrypt(plaintext: string): string {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipheriv(
      this.algorithm,
      Buffer.from(this.encryptionKey, 'hex'),
      iv
    );

    let encrypted = cipher.update(plaintext, 'utf8', 'hex');
    encrypted += cipher.final('hex');

    const authTag = cipher.getAuthTag();
    return `${iv.toString('hex')}.${encrypted}.${authTag.toString('hex')}`;
  }

  static decrypt(encryptedData: string): string {
    const [ivHex, encryptedHex, authTagHex] = encryptedData.split('.');

    const decipher = crypto.createDecipheriv(
      this.algorithm,
      Buffer.from(this.encryptionKey, 'hex'),
      Buffer.from(ivHex, 'hex')
    );

    decipher.setAuthTag(Buffer.from(authTagHex, 'hex'));

    let decrypted = decipher.update(encryptedHex, 'hex', 'utf8');
    decrypted += decipher.final('utf8');

    return decrypted;
  }
}

@Entity('users')
export class User {
  @Column({ type: 'text', select: false })
  email_encrypted: string;

  // 가상 속성
  get email(): string {
    if (this.email_encrypted) {
      return UserEncryption.decrypt(this.email_encrypted);
    }
    return '';
  }

  set email(value: string) {
    this.email_encrypted = UserEncryption.encrypt(value);
  }
}
```

#### 민감 정보 마스킹

```typescript
// backend/api/src/utils/mask.util.ts
export class DataMasking {
  // 이메일 마스킹 (example@domain.com → ex****@domain.com)
  static maskEmail(email: string): string {
    const [name, domain] = email.split('@');
    const maskedName = name.substring(0, 2) + '*'.repeat(name.length - 2);
    return `${maskedName}@${domain}`;
  }

  // 전화번호 마스킹 (010-1234-5678 → 010-****-5678)
  static maskPhone(phone: string): string {
    return phone.replace(/(\d{3})-(\d{4})-(\d{4})/, '$1-****-$3');
  }

  // 신용카드 마스킹 (1234567890123456 → ****5678)
  static maskCard(cardNumber: string): string {
    const cleaned = cardNumber.replace(/\D/g, '');
    return '*'.repeat(cleaned.length - 4) + cleaned.slice(-4);
  }

  // 주민등록번호 마스킹 (123456-7654321 → 123456-****)
  static maskSSN(ssn: string): string {
    return ssn.replace(/(\d{6})-(.*)/, '$1-****');
  }
}

// 로그에서 민감 정보 제거
export class LogSanitizer {
  private static readonly sensitiveKeys = [
    'password', 'token', 'secret', 'apiKey', 'refreshToken',
    'email', 'phone', 'ssn', 'creditCard', 'authorization'
  ];

  static sanitize(data: any): any {
    if (typeof data !== 'object' || data === null) {
      return data;
    }

    const sanitized = Array.isArray(data) ? [...data] : { ...data };

    for (const key in sanitized) {
      // 민감한 키 확인
      if (this.sensitiveKeys.some(sensitive =>
          key.toLowerCase().includes(sensitive.toLowerCase()))) {
        sanitized[key] = '***REDACTED***';
      } else if (typeof sanitized[key] === 'object') {
        sanitized[key] = this.sanitize(sanitized[key]);
      }
    }

    return sanitized;
  }
}

// 사용 예시
app.use((req, res, next) => {
  const originalJson = res.json;

  res.json = function(data) {
    const sanitized = LogSanitizer.sanitize(data);
    console.log('Response:', sanitized); // 안전한 로그
    return originalJson.call(this, data);
  };

  next();
});
```

```typescript
import cors from 'cors';

// 허용할 출처 목록
const allowedOrigins = [
  'https://fans.ai.kr',
  'https://www.fans.ai.kr',
  process.env.NODE_ENV === 'development' ? 'http://localhost:3001' : null,
  process.env.NODE_ENV === 'development' ? 'http://localhost:3000' : null,

// CORS 설정

  // 요청의 출처 검증

      callback(null, true);
    } else {
      callback(new Error('Not allowed by CORS policy'));
    }
  },

  // 자격증명 포함 허용
  credentials: true,

  // 허용할 HTTP 메서드
  methods: ['GET', 'POST', 'PUT', 'DELETE', 'PATCH', 'OPTIONS'],

  // 허용할 헤더
  allowedHeaders: [
    'Content-Type',
    'Authorization',
    'X-Requested-With',
    'Accept',
  ],

  // 노출할 응답 헤더
  exposedHeaders: ['X-Total-Count', 'X-Page-Count'],

  // preflight 캐시 시간 (초)

};
```

#### Rate Limiting

```typescript
// backend/api/src/middleware/rate-limit.middleware.ts
import rateLimit from 'express-rate-limit';
import RedisStore from 'rate-limit-redis';
import redis from './redis';

// 1. 일반 API Rate Limiting (100 req / 15분)
export const apiLimiter = rateLimit({
  store: new RedisStore({
    client: redis,
    prefix: 'rl:api:',
  }),
  windowMs: 15 * 60 * 1000,  // 15분
  max: 100,                   // 최대 100 요청
  message: 'Too many requests from this IP, please try again later.',
  statusCode: 429,
  standardHeaders: true,      // RateLimit-* 헤더 포함
  legacyHeaders: false,       // X-RateLimit-* 헤더 제거
  skip: (req, res) => {
    // 특정 IP는 제외 (내부 서버 등)
    return req.ip === '127.0.0.1';
  },
  keyGenerator: (req, res) => {
    // IP 기반 rate limiting
    return req.ip || req.socket.remoteAddress;
  },
});

// 2. 로그인 API Rate Limiting (5 req / 15분)
export const loginLimiter = rateLimit({
  store: new RedisStore({
    client: redis,
    prefix: 'rl:login:',
  }),
  windowMs: 15 * 60 * 1000,
  max: 5,  // 매우 제한적
  message: 'Too many login attempts, please try again later.',
  statusCode: 429,
  skipSuccessfulRequests: true,  // 성공한 요청은 카운트하지 않음
  skipFailedRequests: false,     // 실패한 요청 카운트
});

// 3. 비밀번호 재설정 Rate Limiting (3 req / 1시간)
export const passwordResetLimiter = rateLimit({
  store: new RedisStore({
    client: redis,
    prefix: 'rl:password:',
  }),
  windowMs: 60 * 60 * 1000,  // 1시간
  max: 3,
  message: 'Too many password reset attempts, please try again later.',
  statusCode: 429,
});

// 4. 검색 API Rate Limiting (30 req / 1분)
export const searchLimiter = rateLimit({
  store: new RedisStore({
    client: redis,
    prefix: 'rl:search:',
  }),
  windowMs: 60 * 1000,  // 1분
  max: 30,
  message: 'Too many search requests, please try again later.',
  statusCode: 429,
});

// 미들웨어 적용
app.use('/api/', apiLimiter);
app.post('/api/auth/login', loginLimiter, loginController);
app.post('/api/auth/password-reset', passwordResetLimiter, passwordResetController);
app.get('/api/search', searchLimiter, searchController);
```

**AWS Secrets Manager 통합**:

```typescript
// backend/api/src/config/secrets.ts
import {
  SecretsManagerClient,
  GetSecretValueCommand,
  ListSecretsCommand
} from '@aws-sdk/client-secrets-manager';

export class SecretsManager {
  private static client = new SecretsManagerClient({
    region: process.env.AWS_REGION || 'ap-northeast-2'
  });

  private static cache = new Map<string, any>();
  private static cacheTTL = 5 * 60 * 1000;  // 5분

  /**
   * 단일 시크릿 조회
   */
  static async getSecret<T = any>(secretName: string): Promise<T> {
    // 캐시 확인
    const cached = this.cache.get(secretName);
    if (cached && cached.expiresAt > Date.now()) {
      return cached.value;
    }

    try {
      const response = await this.client.send(
        new GetSecretValueCommand({ SecretId: secretName })
      );

      const secret = response.SecretString
        ? JSON.parse(response.SecretString)
        : response.SecretBinary;

      // 캐시 저장
      this.cache.set(secretName, {
        value: secret,
        expiresAt: Date.now() + this.cacheTTL
      });

      return secret;
    } catch (error) {
      console.error(`Error retrieving secret ${secretName}:`, error);
      throw new Error(`Failed to retrieve secret: ${secretName}`);
    }
  }

  /**
   * 모든 시크릿 나열
   */
  static async listSecrets(filter?: string): Promise<string[]> {
    try {
      const response = await this.client.send(new ListSecretsCommand({}));

      const secrets = response.SecretList?.map(s => s.Name) || [];

      if (filter) {
        return secrets.filter(s => s?.includes(filter));
      }

      return secrets;
    } catch (error) {
      console.error('Error listing secrets:', error);
      throw new Error('Failed to list secrets');
    }
  }

  /**
   * 시크릿 로드 (앱 시작 시)
   */
  static async loadSecrets(): Promise<void> {
    const environment = process.env.NODE_ENV || 'development';
    const secretNames = [
      `fans/${environment}/api`,
      `fans/${environment}/database`,
      `fans/${environment}/oauth`,
      `fans/${environment}/aws`,
    ];

    for (const secretName of secretNames) {
      try {
        const secret = await this.getSecret(secretName);

        // 환경 변수에 설정
        Object.entries(secret).forEach(([key, value]) => {
          process.env[key] = String(value);
        });

        console.log(`Loaded secrets from ${secretName}`);
      } catch (error) {
        console.warn(`Could not load ${secretName}, using environment variables`);
      }
    }
  }

  /**
   * 캐시 초기화
   */
  static clearCache(secretName?: string): void {
    if (secretName) {
      this.cache.delete(secretName);
    } else {
      this.cache.clear();
    }
  }
}

// 앱 시작 시 호출
import { SecretsManager } from './config/secrets';

app.listen(3000, async () => {
  // 시크릿 로드
  await SecretsManager.loadSecrets();

  console.log('Server running on port 3000');
});
```

```typescript
  'JWT_SECRET',
  'ENCRYPTION_KEY',
];

requiredEnvVars.forEach(envVar => {
  if (!process.env[envVar]) {
    throw new Error(`Missing required environment variable: ${envVar}`);
  }
});

console.log('Environment configuration validated');
```
