def summarize_news(self, title: str, content: str, max_length: int = 100):
    # 1. 텍스트 전처리
    cleaned_title = self.clean_text(title)
    cleaned_content = self.clean_text(content)
    full_text = f"{cleaned_title}. {cleaned_content}"

    # 2. 토큰 길이 제한 (512토큰)

        full_text = full_text[:2000] + "..."

    if len(full_text.strip()) < 50:

        full_text,
        max_length=max_length,      # 100자
        min_length=30,               # 30자
        do_sample=True,
        temperature=0.7
    )

    return {
        "summary": summary_result[0]['summary_text'],

            '생활/문화': ['여행', '맛집', '레시피', '패션', ...],

    def classify(self, title: str, content: str):

        # 각 카테고리별 키워드 매칭 점수 계산
        scores = {}
        for category, keywords in self.category_keywords.items():
            score = sum(1 for keyword in keywords if keyword.lower() in text)
            if score > 0:
                scores[category] = score

        # 최소 2개 이상의 키워드 매칭 필요
        if scores:
            best_category = max(scores, key=scores.get)
            if scores[best_category] >= 2:
                return best_category

        return '기타'

def calculate_bias_score(self, text: str):
    # 여당/야당 키워드 분석
    ruling_score = self.count_keywords(text, self.ruling_keywords)
    opposition_score = self.count_keywords(text, self.opposition_keywords)

    # 감성 분석
    ruling_sentiment = self.sentiment_analyzer.analyze(text, "여당")
    opposition_sentiment = self.sentiment_analyzer.analyze(text, "야당")

    # 최종 편향 점수 계산
    bias_score = (ruling_sentiment - opposition_sentiment) / 2

    if bias_score < -0.5:
        stance = "진보"
    elif bias_score < -0.2:
        stance = "약진보"
    elif bias_score < 0.2:
        stance = "중도"
    elif bias_score < 0.5:
        stance = "약보수"
    else:
        stance = "보수"

    return {
        "bias_score": bias_score,       # -1.0 ~ +1.0
        "stance": stance,               # 진보, 약진보, 중도, 약보수, 보수
        "ruling_sentiment": ruling_sentiment,
        "opposition_sentiment": opposition_sentiment

```typescript
private mergeAndScoreArticles(
  collaborative, contentBased, keywordBased, biasBased, trending
) {
  const articleMap = new Map<number, any>();

  // 각 추천 타입별 점수 합산
  collaborative.forEach(article => {
    articleMap.set(article.id, {
      ...article,
      final_score: article.score * 0.25  // 협업 필터링
    });
  });

  keywordBased.forEach(article => {
    if (articleMap.has(article.id)) {
      articleMap.get(article.id).final_score += article.score * 0.30;
    } else {
      articleMap.set(article.id, {
        ...article,
        final_score: article.score * 0.30
      });
    }
  });

  // ... contentBased (0.20), biasBased (0.15), trending (0.10)

  // 점수순 정렬
  return Array.from(articleMap.values())
    .sort((a, b) => b.final_score - a.final_score);
}
```

기사 B:

- 트렌딩: 6점 * 0.10 = 0.60

router.post('/login', async (req, res) => {
  const { email, password } = req.body;

  // 1. 사용자 조회
  const user = await userRepo.findOne({ where: { email } });

  // 2. 비밀번호 검증 (bcrypt)
  const isValid = await bcrypt.compare(password, user.password_hash);

  if (!isValid) {
    return res.status(401).json({ error: 'INVALID_CREDENTIALS' });
  }

  // 3. JWT 토큰 생성 (authService.ts)
  const token = jwt.sign(
    { userId: user.id, email: user.email },
    process.env.JWT_SECRET,
    { expiresIn: '7d' }
  );

  // 4. 마지막 로그인 시각 업데이트
  await userRepo.update(user.id, { last_login: new Date() });

  res.json({
    token,
    user: {
      id: user.id,
      email: user.email,
      username: user.username
    }
  });
});
```

    }
  });

  if (!user) {
    user = await userRepo.save({
      email: kakaoUser.kakao_account.email,
      username: kakaoUser.kakao_account.profile.nickname,
      password_hash: '',  // 소셜 로그인은 비밀번호 없음
      provider: 'kakao',
      social_token: accessToken,
      profile_image: kakaoUser.kakao_account.profile.profile_image_url
    });
  }

  // 4. JWT 토큰 생성
  const token = jwt.sign(
    { userId: user.id, email: user.email },
    process.env.JWT_SECRET,
    { expiresIn: '7d' }
  );

  // 5. 프론트엔드로 리다이렉트
  res.redirect(`http://localhost:3001/login-success?token=${token}`);
});
```

### 🛡️ JWT 검증 미들웨어 (authMiddleware.ts)

```typescript
export const authMiddleware = (req, res, next) => {
  // 1. Authorization 헤더 확인
  const authHeader = req.headers.authorization;
  if (!authHeader) {
    return res.status(401).json({ error: 'NO_TOKEN' });
  }

  // 2. Bearer 토큰 추출
  const token = authHeader.split(' ')[1];  // "Bearer <token>"

  // 3. JWT 검증
  try {
    const decoded = jwt.verify(token, process.env.JWT_SECRET);
    req.user = decoded;  // { userId, email }
    next();
  } catch (error) {
    return res.status(401).json({ error: 'INVALID_TOKEN' });
  }
};
```
