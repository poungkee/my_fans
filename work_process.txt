  ai-summary.queue.ts 예시:
  import { Queue } from 'bullmq';
  import { redisConnection } from '../config/bullmq';
  import { QUEUE_NAMES, QUEUE_OPTIONS } from '../config/queue-names';
  import { SummaryJob } from '../types/job.types';

  export const aiSummaryQueue = new Queue<SummaryJob>(
    QUEUE_NAMES.AI_SUMMARY,
    {
      connection: redisConnection,
      defaultJobOptions: QUEUE_OPTIONS.defaultJobOptions
    }
  );

  export async function addSummaryJob(data: SummaryJob) {
    return await aiSummaryQueue.add('summarize', data);
  }

  새 코드:
  // ✅ 큐 기반 방식
  import { recommendationQueue } from '@fans/queue';
  import { getRedisClient } from '../config/redis';

  router.get('/personalized/:userId', async (req, res) => {
    const { userId } = req.params;
    const limit = parseInt(req.query.limit as string) || 20;

    const redis = getRedisClient();
    const cacheKey = `recommendations:user:${userId}`;

    // 1. 캐시 확인
    const cached = await redis.get(cacheKey);
    if (cached) {
      return res.json({
        ...JSON.parse(cached),
        source: 'cache'
      });
    }

    // 2. 큐에 작업 추가
    await recommendationQueue.add('generate', {
      userId,
      limit,
      priority: 'high'
    });

    // 3. 임시 인기 기사 반환
    const popularArticles = await getPopularArticles(limit);

    res.json({
      articles: popularArticles,
      status: 'generating',
      message: '맞춤 추천을 생성하고 있습니다. 잠시 후 새로고침해주세요.',
      source: 'fallback'
    });
  });

  // 추천 완성 여부 확인 API (프론트엔드 폴링용)
  router.get('/personalized/:userId/status', async (req, res) => {
    const { userId } = req.params;
    const redis = getRedisClient();

    const cached = await redis.get(`recommendations:user:${userId}`);

    res.json({
      ready: !!cached,
      timestamp: cached ? Date.now() : null
    });
  });
